async function askAI() { const q = $("question").value.trim(); if (!q) return; const chat = $("chatBox"); // Show user question chat.innerHTML += `
${esc(q)}
`; $("question").value = ""; // Thinking message const thinking = document.createElement("div"); thinking.className = "msg bot"; thinking.innerHTML = ` EduPath AI is thinking… `; chat.appendChild(thinking); chat.scrollTop = chat.scrollHeight; try { // ----------------------------------------- // GET CURRENT LEARNER INFORMATION // ----------------------------------------- const boxes = [ ...document.querySelectorAll( "#weeklyPlan input[type=checkbox]" ) ]; const completed = boxes.filter(x => x.checked).length; const total = boxes.length; const progressPercent = total ? Math.round((completed / total) * 100) : 0; const career = state.profile?.career || "the learner's target career"; const experience = state.profile?.experience || "Beginner"; const goal = state.profile?.goal || "Become job-ready"; const skills = state.profile?.aiSummary?.extracted_skills || ( state.profile?.skills ? state.profile.skills .split(",") .map(x => x.trim()) .filter(Boolean) : [] ); const gaps = state.analysis?.gaps || []; const remainingTasks = boxes .filter(x => !x.checked) .map(x => x.parentElement.innerText) .slice(0, 5); // ----------------------------------------- // LIVE GEMINI AI // ----------------------------------------- if ($("apiKey").value.trim()) { const context = { learner: { target_career: career, experience_level: experience, goal: goal, current_skills: skills }, skill_gaps: gaps.map(g => ({ skill: g.skill, current_level: g.current_level, required_level: g.required_level, priority: g.priority, reason: g.reason })), progress: { completed_activities: completed, total_activities: total, percentage: progressPercent }, remaining_tasks: remainingTasks, user_question: q }; const prompt = ` You are EduPath AI, a personalized learning coach. You MUST answer the user's EXACT question. Do NOT give the same generic answer for different questions. The learner may ask about: - a specific skill - a career - programming - SQL - Python - projects - interviews - learning resources - skill gaps - difficult topics - their progress - what to learn next - why something is important - how to learn something - any other learning-related question IMPORTANT RULES: 1. First understand the exact question. 2. Answer that question directly. 3. Do not simply repeat the learner profile. 4. Do not always recommend the first skill gap. 5. Do not always give the same roadmap. 6. If the learner asks about a specific technology, explain that technology. 7. If the learner asks "why", explain WHY that particular thing matters. 8. If the learner asks "how", give step-by-step instructions. 9. If the learner asks for a project, give a specific project. 10. If the learner asks about a career, explain the relevant skills. 11. If the learner says they are struggling, identify the prerequisite they may be missing. 12. Use the learner's current skills and progress when relevant. 13. Do not invent information about the learner. 14. Keep the response practical and beginner-friendly. 15. Do not answer a different question just because it matches a keyword. 16. The user's question is more important than generic roadmap rules. LEARNER INFORMATION: ${JSON.stringify(context, null, 2)} USER'S EXACT QUESTION: "${q}" Before answering, internally determine: - What exactly is the user asking? - Is it asking WHAT, WHY, HOW, WHERE, WHEN, or something else? - Which skill/topic is being discussed? - Does the learner's profile affect the answer? Then answer specifically. Return ONLY valid JSON in this format: { "answer": "Direct answer to the user's exact question.", "next_actions": [ "specific action 1", "specific action 2", "specific action 3" ], "practice": "One practical exercise related to the question.", "next_step": "The single most useful immediate next step." } `; const result = await gemini( prompt, null, ` You are the Adaptive Coach Agent of EduPath AI. Your job is to have a real conversation with the learner. Every question must be analyzed independently. Never reuse a generic response merely because the question contains words such as "why", "how", "project", "gap", or "learn". Answer the exact question asked by the learner. Make the response personalized, specific and practical. ` ); // ----------------------------------------- // BUILD AI RESPONSE // ----------------------------------------- let output = ""; if (result.answer) { output += result.answer; } if ( Array.isArray(result.next_actions) && result.next_actions.length ) { output += "\n\nNEXT ACTIONS:\n"; output += result.next_actions .map((action, index) => `${index + 1}. ${action}` ) .join("\n"); } if (result.practice) { output += `\n\nPRACTICE:\n${result.practice}`; } if (result.next_step) { output += `\n\nYOUR NEXT STEP:\n${result.next_step}`; } thinking.textContent = output || "I couldn't generate an answer."; } // ----------------------------------------- // OFFLINE MODE // ----------------------------------------- else { const lower = q.toLowerCase(); let answer = ""; let actions = []; let practice = ""; // Specific skill questions if ( lower.includes("python") || lower.includes("sql") || lower.includes("spark") || lower.includes("javascript") || lower.includes("java") || lower.includes("cloud") ) { let topic = "this technology"; if (lower.includes("python")) topic = "Python"; else if (lower.includes("sql")) topic = "SQL"; else if (lower.includes("spark")) topic = "Apache Spark"; else if (lower.includes("javascript")) topic = "JavaScript"; else if (lower.includes("java")) topic = "Java"; else if (lower.includes("cloud")) topic = "Cloud"; answer = `For your ${career} goal, ${topic} should be learned through practical use rather than only theory.`; actions = [ `Learn the fundamentals of ${topic}`, `Practice 3-5 small problems`, `Build one small ${career}-related example` ]; practice = `Create a small ${career}-related task using ${topic}.`; } // What should I learn? else if ( lower.includes("what should i learn") || lower.includes("learn next") || lower.includes("next skill") ) { const firstGap = gaps[0]; if (firstGap) { answer = `Based on your current ${career} roadmap, your next learning target is ${firstGap.skill}. Your current level is ${firstGap.current_level}, while the target level is ${firstGap.required_level}.`; actions = [ `Learn the fundamentals of ${firstGap.skill}`, `Complete 3-5 practical exercises`, `Build a small project using ${firstGap.skill}` ]; practice = `Create one small ${career} task using ${firstGap.skill}.`; } } // Project else if ( lower.includes("project") || lower.includes("portfolio") ) { answer = `For a ${career} portfolio, build a project that solves a realistic problem and demonstrates the skills you are currently developing.`; actions = [ "Choose a real-world problem", "Select the required technologies", "Build a working version", "Test the project", "Document it on GitHub" ]; practice = `Build a small end-to-end project related to ${career}.`; } // Interview else if ( lower.includes("interview") ) { answer = `For a ${career} interview, prepare technical fundamentals, problem-solving questions and clear explanations of your projects.`; actions = [ "Review important technical concepts", "Solve role-specific problems", "Prepare explanations for your projects", "Practice interview questions" ]; practice = `Explain one project using: Problem → Approach → Technology → Result → Improvements.`; } // Progress else if ( lower.includes("progress") || lower.includes("completed") ) { answer = `You have completed ${completed} of ${total} planned activities, which is ${progressPercent}% of your current learning plan.`; actions = remainingTasks.length ? remainingTasks.slice(0, 3) : [ "Review completed topics", "Build a portfolio project", "Start interview preparation" ]; practice = "Complete your next unfinished activity without switching topics."; } // How questions else if ( lower.startsWith("how ") || lower.includes("how can i") || lower.includes("how do i") ) { answer = `To answer "${q}", break the problem into small steps and practice each step with a real example related to ${career}.`; actions = [ "Understand the basic concept", "Follow one simple example", "Try the example yourself", "Apply it to a small project" ]; practice = `Try solving one small ${career}-related problem using the concept you are asking about.`; } // Why questions else if ( lower.startsWith("why ") || lower.includes("why is") || lower.includes("why should") ) { answer = `The reason depends on the specific topic you are asking about. For your ${career} goal, connect the concept to a real task so you can see where it is useful.`; actions = [ "Identify what the concept is used for", "Look at one real-world example", "Try a small practical exercise" ]; practice = `Find one ${career}-related problem where this concept could be useful.`; } // Default else { answer = `Your question is about: "${q}" For your ${career} goal, the best approach is to connect this topic with your current skills and learning gaps.`; actions = [ "Break the topic into smaller concepts", "Learn the fundamentals", "Practice with an example" ]; practice = `Try explaining "${q}" in your own words and create one small example.`; } // ----------------------------------------- // FORMAT OFFLINE RESPONSE // ----------------------------------------- let output = answer; if (actions.length) { output += "\n\nNEXT ACTIONS:\n"; output += actions .map((x, i) => `${i + 1}. ${x}`) .join("\n"); } if (practice) { output += `\n\nPRACTICE:\n${practice}`; } thinking.textContent = output; } } catch (e) { console.error("EduPath AI Error:", e); thinking.textContent = `I couldn't connect to the live AI agent. Please check: 1. Your Gemini API key 2. Your internet connection 3. The Gemini model/API configuration You can also remove the API key and use EduPath's offline mode.`; } chat.scrollTop = chat.scrollHeight; }
